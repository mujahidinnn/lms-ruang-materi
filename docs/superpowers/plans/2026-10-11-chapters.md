# Chapters Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn every topic into a course of readable chapters (markdown lesson, code examples, slide embeds, a short quiz each) with per-chapter progress, AI-drafted from the deck and editable by admins.

**Architecture:** Three new tables (`chapters`, `chapter_questions`, `chapter_progress`) follow the existing draft/publish content pattern and RLS. Two security-definer RPCs (`open_chapter`, `complete_chapter`) write learner state. Public pages stay cached (`'use cache'` + `cacheTag('content')`); per-learner state loads client-side. Markdown renders with `react-markdown` (no raw HTML) and `shiki` for code.

**Tech Stack:** Next.js 16 (App Router, cacheComponents), Supabase Postgres + RLS, pgTAP, Vitest, Playwright, Tailwind v4, zod, react-markdown 10, remark-gfm 4, shiki 4.

**Spec:** `docs/superpowers/specs/2026-10-11-chapters-design.md`

## Global Constraints

- UI text in Bahasa Indonesia, code comments in English (GUIDELINE "Style").
- Pastel style only through `components/ui/styles.ts` (`primaryButton`, `secondaryButton`, `card`, `chip`, `mutedChip`, `tile(i)`) and zinc/tile/accent/brand tokens; no hex, no `white`/`black`.
- Every learner write goes through a database function or RLS; never trust the client for completion.
- Chapter `slug` matches `^[a-z0-9-]+$` and is never `slide` (route `/belajar/[slug]/slide`).
- 4 to 8 chapters per AI draft, 400 to 800 words each, 2 to 4 quiz questions each, `est_minutes` 1 to 60.
- Migrations are tested on the local stack (`npx supabase db reset && npx supabase test db`) before `npm run db:push`.
- Commit after each task; commits end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

Deliberate deviations from the spec (simpler, same behaviour):
- `chapter_questions` has no `status`; it is visible when its chapter is.
- Opening a chapter is recorded by a new `open_chapter()` RPC instead of a new `touch_progress` parameter (an overload would make the 3-argument RPC call ambiguous).
- Adding new chapter quiz questions in admin is out of scope; admins edit or delete the drafted ones.

## Review Focus

- A topic published before re-import has zero chapters: `/belajar/[slug]` must render today's page, and `generateStaticParams` for `/belajar/[slug]/[bab]` must still return one param (cacheComponents rejects an empty list).
- Republishing a topic (re-import + publish) must keep learners' completed chapters and their resume chapter when the slug is unchanged.
- A chapter with no quiz questions (admin-made) must still be completable ("Tandai bab selesai").
- `![x](slide:99)` past the deck, or `slide:abc`, must render nothing, not a broken image or a crash.
- A guest finishing a chapter quiz must see the result without errors and no write attempt.

-- pattern as the other content tables; chapter_questions follow their
-- chapter's status. Learner state lives in chapter_progress and
-- progress.last_chapter_id, written only by functions (next migration).

create table public.chapters (
  id uuid primary key default gen_random_uuid(),
  topic_id uuid not null references public.topics (id) on delete cascade,
  position int not null check (position >= 0),
  slug text not null check (slug ~ '^[a-z0-9-]+$' and slug <> 'slide'),
  title text not null check (char_length(title) between 1 and 120),
  body text not null default '',
  slide_from int check (slide_from >= 1),
  slide_to int check (slide_to >= 1),
  est_minutes int not null default 10 check (est_minutes between 1 and 60),
  status public.content_status not null default 'draft',
  updated_by uuid references auth.users (id) on delete set null,
  created_at timestamptz not null default now(),
  check (slide_from is null or slide_to is null or slide_from <= slide_to),
  -- Deferred so reordering can swap two positions in one transaction.
  unique (topic_id, status, position) deferrable initially deferred,
  unique (topic_id, status, slug) deferrable initially deferred
);

create table public.chapter_questions (
  id uuid primary key default gen_random_uuid(),

-- Learner writes for chapters, and chapters in publish_topic.

-- Marks the topic sedang (if new) and remembers the chapter for Dasbor.
create or replace function public.open_chapter(p_chapter uuid)
returns void
language plpgsql
security definer
set search_path = ''
as $$
declare
  uid uuid := auth.uid();
  c public.chapters;
begin
  if uid is null then
    raise exception 'Masuk dulu' using errcode = '42501';
  end if;
  select * into c from public.chapters ch
  where ch.id = p_chapter and ch.status = 'published'
    and exists (select 1 from public.topics t where t.id = ch.topic_id and t.status = 'published');
  if not found then
    raise exception 'Bab tidak ditemukan' using errcode = 'P0002';
  end if;
  insert into public.progress (user_id, topic_id, state, last_chapter_id)
  values (uid, c.topic_id, 'sedang', c.id)
  on conflict (user_id, topic_id) do update set last_chapter_id = excluded.last_chapter_id;
end;
$$;

-- Progress first, so the selesai trigger below finds the row it updates.
create or replace function public.complete_chapter(p_chapter uuid)
returns void
language plpgsql
security definer
set search_path = ''
as $$
begin
  perform public.open_chapter(p_chapter);
  insert into public.chapter_progress (user_id, chapter_id)
  values (auth.uid(), p_chapter)
  on conflict do nothing;
end;
$$;

revoke execute on function public.open_chapter(uuid), public.complete_chapter(uuid) from public, anon;
grant execute on function public.open_chapter(uuid), public.complete_chapter(uuid) to authenticated;

-- A topic without a published exam is selesai once every published
-- chapter is done. Topics with an exam still finish only by passing it.
create or replace function public.finish_topic_by_chapters()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
declare
  tid uuid;
begin
  select topic_id into tid from public.chapters where id = new.chapter_id;
  if not exists (select 1 from public.exams where topic_id = tid and status = 'published')
    and not exists (
      select 1 from public.chapters c
      where c.topic_id = tid and c.status = 'published'
        and not exists (select 1 from public.chapter_progress p where p.chapter_id = c.id and p.user_id = new.user_id))
  then
    update public.progress set state = 'selesai' where user_id = new.user_id and topic_id = tid;
  end if;
  return new;
end;
$$;

create trigger finish_topic_by_chapters after insert on public.chapter_progress
  for each row execute function public.finish_topic_by_chapters();

-- Republishing replaces published chapters with drafts. Progress and the
-- resume pointer move to the draft chapter with the same slug.
create or replace function public.carry_chapter_progress()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
declare
  successor uuid;
begin
  if old.status = 'published' then
    select id into successor from public.chapters
    where topic_id = old.topic_id and status = 'draft' and slug = old.slug;
    if successor is not null then
      update public.chapter_progress set chapter_id = successor where chapter_id = old.id;
      update public.progress set last_chapter_id = successor where last_chapter_id = old.id;
    end if;
  end if;
  return old;
end;
$$;

create trigger carry_chapter_progress before delete on public.chapters
  for each row execute function public.carry_chapter_progress();
```

import type { Slide } from "@/lib/slides";

// Pure helpers for chapters: slide embeds, quiz completion, navigation.

// `![keterangan](slide:12)` in a chapter body embeds slide 12.
export function parseSlideRef(src: string): number | null {
  const m = /^slide:(\d+)$/.exec(src);
  const n = m ? Number(m[1]) : 0;
  return n >= 1 ? n : null;
}

export function resolveSlideRef(src: string, slides: Slide[]): Slide | null {
  const n = parseSlideRef(src);
  return n === null ? null : slides.find((s) => s.index === n) ?? null;
}

// Done when every question is answered correctly. A chapter without a quiz
// is finished with its own button instead.
export function quizDone(questions: { id: string; answer: number }[], answers: Record<string, number>): boolean {
  return questions.length > 0 && questions.every((q) => answers[q.id] === q.answer);
}

export function neighbors<T extends { slug: string }>(list: T[], slug: string): { prev: T | null; next: T | null } {
  const i = list.findIndex((c) => c.slug === slug);
  return { prev: i > 0 ? list[i - 1] : null, next: i >= 0 && i < list.length - 1 ? list[i + 1] : null };
}

export function totalMinutes(list: { estMinutes: number }[]): number {
  return list.reduce((sum, c) => sum + c.estMinutes, 0);
}
```

```ts
type ChapterRow = {
  id: string; slug: string; title: string; position: number; est_minutes: number;
  body: string; slide_from: number | null; slide_to: number | null;
  chapter_questions: (ChapterQuestion & { position: number })[];
};

export async function getChapter(topicSlug: string, chapterSlug: string): Promise<Chapter | null> {
  "use cache";
  cacheTag("content");

  const { data, error } = await publicClient
    .from("chapters")
    .select("id, slug, title, position, est_minutes, body, slide_from, slide_to, chapter_questions(id, position, type, prompt, code, options, answer, explanation), topics!inner(slug)")
    .eq("topics.slug", topicSlug)
    .eq("slug", chapterSlug)
    .maybeSingle();
  if (error) throw new Error(`getChapter: ${error.message}`);
  if (!data) return null;
  const c = data as unknown as ChapterRow;
  return {
    id: c.id,
    slug: c.slug,
    title: c.title,
    position: c.position,
    estMinutes: c.est_minutes,
    body: c.body,
    slideFrom: c.slide_from,
    slideTo: c.slide_to,
    questions: c.chapter_questions.toSorted((a, b) => a.position - b.position).map(({ position: _p, ...q }) => q),
  };
}
```

import remarkGfm from "remark-gfm";
import { resolveSlideRef } from "@/lib/chapters";
import { mediumSrc, type Slide } from "@/lib/slides";

// Chapter bodies are markdown from AI drafts or admins: raw HTML is never

export default function ChapterMarkdown({ body, slides, highlight = true }: { body: string; slides: Slide[]; highlight?: boolean }) {
  return (

        remarkPlugins={[remarkGfm]}
        urlTransform={(url) => (url.startsWith("slide:") ? url : defaultUrlTransform(url))}
        components={{
          img: ({ src, alt }) => {
            const slide = typeof src === "string" ? resolveSlideRef(src, slides) : null;
            if (!slide) return null;
            return (
              <a href={slide.src} target="_blank" rel="noopener" className="my-5 block overflow-hidden rounded-2xl shadow-soft">

              </a>
            );
          },
          pre: ({ children }) => <>{children}</>,
          code: ({ className, children }) => {
            const text = String(children ?? "");
            const lang = /language-(\w+)/.exec(className ?? "")?.[1];
            const block = lang !== undefined || text.includes("\n");
            if (!block) return <code className="rounded-md bg-zinc-800 px-1.5 py-0.5 font-mono text-[0.9em]">{children}</code>;
            const code = text.replace(/\n$/, "");
            return highlight ? (

            );
          },
          a: ({ href, children }) => (
            <a href={href} target="_blank" rel="noopener noreferrer" className="text-accent underline underline-offset-2">{children}</a>

"use client";

import { useCallback, useEffect, useState } from "react";
import { createClient } from "@/lib/supabase/client";

// The signed-in learner's finished chapters and resume chapter for one
// topic, read with their own session and filtered to their user id.
export function useChapterProgress(topicId: string, chapterIds: string[]) {
  const [signedIn, setSignedIn] = useState(false);
  const [done, setDone] = useState<Set<string>>(new Set());
  const [lastChapterId, setLast] = useState<string | null>(null);
  const key = chapterIds.join(",");

  useEffect(() => {
    const db = createClient();
    db.auth.getSession().then(async ({ data }) => {
      const uid = data.session?.user.id;
      if (!uid) return;
      const [{ data: rows }, { data: p }] = await Promise.all([
        db.from("chapter_progress").select("chapter_id").eq("user_id", uid).in("chapter_id", key.split(",")),
        db.from("progress").select("last_chapter_id").eq("user_id", uid).eq("topic_id", topicId).maybeSingle(),
      ]);
      setSignedIn(true);
      setDone(new Set((rows ?? []).map((r) => r.chapter_id as string)));
      setLast((p?.last_chapter_id as string | null) ?? null);
    });
  }, [topicId, key]);

  const markDone = useCallback((id: string) => setDone((d) => new Set(d).add(id)), []);
  return { signedIn, done, lastChapterId, markDone };
}

"use client";

import { Check, PartyPopper, X } from "lucide-react";

import { primaryButton, secondaryButton, tile } from "@/components/ui/styles";

// Instant feedback, retry allowed. All correct (or the button, for a
// chapter without questions) completes the chapter for a signed-in learner.
export default function ChapterQuiz({ chapterId, slug, questions: raw }: { chapterId: string; slug: string; questions: ChapterQuestion[] }) {
  const questions = useMemo(() => raw.map(shuffleOptions), [raw]);
  const [answers, setAnswers] = useState<Record<string, number>>({});
  const [status, setStatus] = useState<"idle" | "saved" | "guest" | "error">("idle");

  useEffect(() => {
    // Record the chapter as the resume point; guests make no request.
    const db = createClient();
    db.auth.getSession().then(({ data }) => data.session && db.rpc("open_chapter", { p_chapter: chapterId }).then());
  }, [chapterId]);

  async function complete() {
    const db = createClient();
    const { data } = await db.auth.getSession();
    if (!data.session) return setStatus("guest");
    const { error } = await db.rpc("complete_chapter", { p_chapter: chapterId });
    setStatus(error ? "error" : "saved");
  }

  const finished = quizDone(questions, answers);
  useEffect(() => {
    if (finished && status === "idle") complete();
    // eslint-disable-next-line react-hooks/exhaustive-deps -- complete once when the quiz turns done
  }, [finished]);

  return (
    <section aria-labelledby="kuis" className="mt-12 rounded-[28px] bg-zinc-800/50 p-5 sm:p-8">
      <h2 id="kuis" className="text-2xl font-bold tracking-tight">Cek pemahaman</h2>

                  {q.options.map((opt, n) => {
                    const chosen = picked === n;
                    const right = chosen && n === q.answer;
                    const wrong = chosen && n !== q.answer;
                    return (
                      <button
                        key={n}
                        type="button"
                        aria-pressed={chosen}
                        onClick={() => setAnswers((a) => ({ ...a, [q.id]: n }))}
                        className={`flex min-h-12 items-center gap-3 rounded-2xl px-3 py-2 text-left shadow-soft focus-visible:outline-2 focus-visible:outline-accent ${right ? "bg-tile-mint" : wrong ? "bg-tile-pink" : "bg-zinc-900 hover:bg-zinc-800"}`}

              </li>
            );
          })}
        </ol>
      )}
      <div aria-live="polite" className="mt-5 empty:hidden">

  );
}

export async function simpanBab(form: FormData) {
  await requireAdmin();
  const f = z
    .object({
      id: z.uuid(),
      topic: z.string(),
      title: z.string().trim().min(1).max(120),
      slug,
      body: z.string(),

// New chapters join the set an admin sees: drafts when the topic has draft
// chapters (published with the next Terbitkan), else live right away.
export async function tambahBab(form: FormData) {
  await requireAdmin();
  const { topic } = z.object({ topic: z.string() }).parse(Object.fromEntries(form));
  const db = await createClient();
  const { data: t } = await db.from("topics").select("id, chapters(status, position)").eq("slug", topic).single();
  if (!t) throw new Error("topik tidak ditemukan");
  const rows = t.chapters as { status: string; position: number }[];
  const status = rows.some((c) => c.status === "draft") ? "draft" : "published";
  const position = Math.max(-1, ...rows.filter((c) => c.status === status).map((c) => c.position)) + 1;
  const { error } = await db.from("chapters").insert({
    topic_id: t.id, status, position, slug: `bab-${position + 1}`, title: `Bab ${position + 1}`, body: "Tulis isi bab di sini.",
  });
  if (error) throw new Error(error.message);
  done(topic);
}

export async function hapusBab(form: FormData) {
  await requireAdmin();
  const { id, topic } = z.object({ id: z.uuid(), topic: z.string() }).parse(Object.fromEntries(form));
  const { error } = await (await createClient()).from("chapters").delete().eq("id", id);
  if (error) throw new Error(error.message);
  done(topic);
}

// Swaps positions with the neighbour; the unique key is deferred.
export async function geserBab(form: FormData) {
  await requireAdmin();
  const { id, topic, arah } = z.object({ id: z.uuid(), topic: z.string(), arah: z.enum(["naik", "turun"]) }).parse(Object.fromEntries(form));
  const db = await createClient();
  const { data: c } = await db.from("chapters").select("topic_id, status, position").eq("id", id).single();
  if (!c) throw new Error("bab tidak ditemukan");
  const { data: other } = await db
    .from("chapters")
    .select("id, position")
    .eq("topic_id", c.topic_id)
    .eq("status", c.status)
    .order("position", { ascending: arah === "turun" })
    .filter("position", arah === "naik" ? "lt" : "gt", c.position)
    .limit(1)
    .maybeSingle();
  if (!other) return;
  const { error } = await db.rpc("swap_chapters", { a: id, b: other.id });
  if (error) throw new Error(error.message);
  done(topic);
}

language plpgsql
security invoker

declare
  pa int;
  pb int;

  select position into pa from public.chapters where id = a;
